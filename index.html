typedef enum {
    DIST_HSL,
    DIST_RGB
} EnumDistance;

static int distance256(int table256[256][3], int i, int a, int b, int c)
{
    int da = a - table256[i][0];
    int db = b - table256[i][1];
    int dc = c - table256[i][2];
    return da*da + db*db + dc*dc;
}

static int find_ansi_index(int table256[256][3], int a, int b, int c)
{
    int index = 0;
    for (int i = 0; i < 256; ++i) {
        if (distance256(table256, i, a, b, c) < distance256(table256, index, a, b, c)) {
            index = i;
        }
    }
    return index;
}

class LineCounter(Canvas):
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.text_widget = None
        
    def attach_widget(self, text_widget):
        self.text_widget = text_widget

    def update(self, *args):
        self.delete("all")
        index = self.text_widget.index("@0,0")
        while True:
            line = self.text_widget.dlineinfo(index)
            if line is None: 
                break
            pos = line[1]
            number = str(index).split(".")[0]
            self.create_text(10, pos, anchor="nw", text=number, fill="white")
            index = self.text_widget.index("%s+1line" % index)

class TextWidget(Text):
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        
        self.insert(1.0, START_TEXT)

        self._origin = self._w + "_orig"
        self.tk.call("rename", self._w, self._origin)
        self.tk.createcommand(self._w, self._proxy)
        
    def _proxy(self, *args):
        cmd = (self._origin,) + args
        result = self.tk.call(cmd)

        if (args[0] in ("insert", "replace", "delete") or
            args[0:3] == ("mark", "set", "insert") or
            args[0:2] == ("xview", "moveto") or
            args[0:2] == ("xview", "scroll") or
            args[0:2] == ("yview", "moveto") or
            args[0:2] == ("yview", "scroll") 
        ):
            self.event_generate("<>", when="tail")

        return result
